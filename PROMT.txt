# bot/proxy_session.py
"""
Кастомная сессия aiogram, которая гоняет все запросы к Telegram Bot API
через твой Cloudflare Worker-прокси.
"""
from aiogram.client.session.aiohttp import AiohttpSession
from aiogram.client.telegram import TelegramAPIServer

PROXY_BASE = "https://tg-proxy.lammusiic01.workers.dev"


def build_proxy_session() -> AiohttpSession:
    """
    Пересылаем запросы к api.telegram.org через воркер.
    Формат вызова у Bot API:
      https://api.telegram.org/bot<token>/<method>
      https://api.telegram.org/file/bot<token>/<path>
    aiogram позволяет подменить базовые URL через TelegramAPIServer.
    """
    api_server = TelegramAPIServer(
        base=f"{PROXY_BASE}/bot{{token}}/{{method}}",
        file=f"{PROXY_BASE}/file/bot{{token}}/{{path}}",
    )
    return AiohttpSession(api=api_server)
⚠️ Важно: твой Cloudflare Worker должен принимать путь после / и форвардить его на https://api.telegram.org/... с теми же методом, телом и заголовками. Если воркер уже так работает — всё заведётся без изменений.

# bot/main.py
import asyncio
import logging
from aiogram import Bot, Dispatcher
from aiogram.enums import ParseMode
from aiogram.client.default import DefaultBotProperties

from core.config import settings
from core.database import init_db
from bot.proxy_session import build_proxy_session
from bot.handlers import start, payment

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
)
logger = logging.getLogger(__name__)


async def main():
    await init_db()

    bot = Bot(
        token=settings.BOT_TOKEN,
        session=build_proxy_session(),
        default=DefaultBotProperties(parse_mode=ParseMode.HTML),
    )
    dp = Dispatcher()
    dp.include_router(start.router)
    dp.include_router(payment.router)

    me = await bot.get_me()
    logger.info(f"🚀 Бот @{me.username} запущен через прокси")

    try:
        await dp.start_polling(bot)
    finally:
        await bot.session.close()


if __name__ == "__main__":
    asyncio.run(main())

# engine/bot_detector.py
"""
Детектор бот-трафика на чистой статистике.
НЕ использует AI — только математика и правила.
"""
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional


@dataclass
class BotPattern:
    code: str
    severity: str          # low / medium / high / critical
    title: str
    description: str
    confidence: float      # 0.0 - 1.0
    evidence: dict = field(default_factory=dict)


@dataclass
class BotReport:
    period_from: str
    period_to: str
    total_clicks: int
    total_cost: float
    bot_clicks_estimate: int
    bot_cost_estimate: float
    bot_percentage: float
    patterns: list[BotPattern]
    actions: list[dict]     # Готовые действия для исполнения скриптом


class BotDetector:
    # Пороги подобраны на основе индустриальных данных
    THRESHOLDS = {
        "min_session_sec": 5,
        "critical_session_sec": 2,
        "max_bounce_rate": 90,
        "critical_bounce_rate": 98,
        "min_pages_per_visit": 1.2,
        "night_hours_start": 1,
        "night_hours_end": 6,
        "max_night_traffic_ratio": 0.25,
        "max_single_city_ratio": 0.35,
        "robot_pct_warning": 10,
        "robot_pct_critical": 25,
    }

    def analyze(
        self,
        metrika_robot_stats: dict,
        metrika_suspicious: dict,
        metrika_geo: dict,
        metrika_hourly: dict,
        direct_stats: list[dict],
        period_from: str,
        period_to: str,
    ) -> BotReport:
        patterns: list[BotPattern] = []
        actions: list[dict] = []

        total_clicks, total_cost = self._sum_direct_stats(direct_stats)

        # ─── 1. Процент роботов по Метрике ─────────────────
        robot_pct = self._extract_robot_pct(metrika_robot_stats)
        if robot_pct >= self.THRESHOLDS["robot_pct_critical"]:
            patterns.append(BotPattern(
                code="high_robot_pct",
                severity="critical",
                title="Критический уровень роботов",
                description=f"Метрика определяет {robot_pct:.1f}% трафика как роботов",
                confidence=0.95,
                evidence={"robot_pct": robot_pct},
            ))
        elif robot_pct >= self.THRESHOLDS["robot_pct_warning"]:
            patterns.append(BotPattern(
                code="elevated_robot_pct",
                severity="high",
                title="Повышенный уровень роботов",
                description=f"{robot_pct:.1f}% трафика — роботы",
                confidence=0.85,
                evidence={"robot_pct": robot_pct},
            ))

        # ─── 2. Короткие сессии + высокий bounce ────────────
        short = self._count_short_sessions(metrika_suspicious)
        if short["ratio"] > 0.3:
            patterns.append(BotPattern(
                code="short_sessions",
                severity="high" if short["ratio"] > 0.5 else "medium",
                title="Массовые короткие сессии",
                description=(
                    f"{short['ratio']*100:.1f}% визитов короче "
                    f"{self.THRESHOLDS['min_session_sec']} сек "
                    f"с отказом > {self.THRESHOLDS['max_bounce_rate']}%"
                ),
                confidence=0.9,
                evidence=short,
            ))
            actions.append({
                "type": "raise_min_session_filter",
                "params": {"min_sec": 5},
                "reason": "Отсечь явно ботовый трафик по длительности",
            })

        # ─── 3. Географические аномалии ────────────────────
        geo_anomalies = self._detect_geo_anomalies(metrika_geo)
        for anom in geo_anomalies[:5]:
            patterns.append(BotPattern(
                code="geo_anomaly",
                severity="medium",
                title=f"Аномалия: {anom['city']}",
                description=(
                    f"{anom['visits']} визитов, "
                    f"отказы {anom['bounce_rate']:.0f}%, "
                    f"сессия {anom['avg_duration']:.1f} сек"
                ),
                confidence=0.75,
                evidence=anom,
            ))
        if geo_anomalies:
            actions.append({
                "type": "exclude_regions",
                "params": {"cities": [a["city"] for a in geo_anomalies[:3]]},
                "reason": "Исключить регионы с бот-фермами",
            })

        # ─── 4. Ночной трафик ──────────────────────────────
        night_ratio = self._calc_night_ratio(metrika_hourly)
        if night_ratio > self.THRESHOLDS["max_night_traffic_ratio"]:
            patterns.append(BotPattern(
                code="night_traffic_spike",
                severity="medium",
                title="Аномальный ночной трафик",
                description=(
                    f"{night_ratio*100:.1f}% трафика идёт ночью "
                    f"(1:00–6:00), это характерно для ботов"
                ),
                confidence=0.7,
                evidence={"night_ratio": night_ratio},
            ))
            actions.append({
                "type": "disable_night_schedule",
                "params": {"from_hour": 1, "to_hour": 6},
                "reason": "Отключить показы в часы бот-активности",
            })

        # ─── Итоговая оценка бот-доли ──────────────────────
        bot_ratio = self._compute_bot_ratio(robot_pct, short, night_ratio)
        bot_clicks = int(total_clicks * bot_ratio)
        bot_cost = round(total_cost * bot_ratio, 2)

        return BotReport(
            period_from=period_from,
            period_to=period_to,
            total_clicks=total_clicks,
            total_cost=total_cost,
            bot_clicks_estimate=bot_clicks,
            bot_cost_estimate=bot_cost,
            bot_percentage=round(bot_ratio * 100, 2),
            patterns=patterns,
            actions=actions,
        )

    # ─── Вспомогательные методы ─────────────────────────────
    def _sum_direct_stats(self, stats: list[dict]) -> tuple[int, float]:
        clicks, cost = 0, 0.0
        for row in stats:
            try:
                clicks += int(row.get("Clicks", 0))
                cost += float(row.get("Cost", 0)) / 1_000_000
            except (TypeError, ValueError):
                pass
        return clicks, round(cost, 2)

    def _extract_robot_pct(self, data: dict) -> float:
        try:
            for row in data.get("data", []):
                dims = row.get("dimensions", [])
                if dims and dims[0].get("name") in ("Yes", "Да"):
                    return float(row["metrics"][1]) if len(row["metrics"]) > 1 else 0.0
        except Exception:
            pass
        return 0.0

    def _count_short_sessions(self, data: dict) -> dict:
        total, suspicious = 0, 0
        for row in data.get("data", []):
            try:
                m = row.get("metrics", [0, 0, 0])
                visits = int(m[0]); bounce = float(m[1]); dur = float(m[2])
                total += visits
                if (bounce > self.THRESHOLDS["max_bounce_rate"]
                        and dur < self.THRESHOLDS["min_session_sec"]):
                    suspicious += visits
            except Exception:
                continue
        return {
            "total": total,
            "suspicious": suspicious,
            "ratio": (suspicious / total) if total else 0.0,
        }

    def _detect_geo_anomalies(self, data: dict) -> list[dict]:
        out = []
        for row in data.get("data", []):
            try:
                city = row["dimensions"][0].get("name", "—")
                m = row.get("metrics", [0, 0, 0])
                visits = int(m[0]); bounce = float(m[1]); dur = float(m[2])
                if visits >= 20 and bounce >= 90 and dur < 5:
                    out.append({
                        "city": city,
                        "visits": visits,
                        "bounce_rate": bounce,
                        "avg_duration": dur,
                    })
            except Exception:
                continue
        return sorted(out, key=lambda x: x["visits"], reverse=True)

    def _calc_night_ratio(self, hourly: dict) -> float:
        total, night = 0, 0
        start = self.THRESHOLDS["night_hours_start"]
        end = self.THRESHOLDS["night_hours_end"]
        for row in hourly.get("data", []):
            try:
                hour = int(row["dimensions"][0].get("name", -1))
                visits = int(row["metrics"][0])
                total += visits
                if start <= hour <= end:
                    night += visits
            except Exception:
                continue
        return (night / total) if total else 0.0

    def _compute_bot_ratio(self, robot_pct: float, short: dict, night: float) -> float:
        """
        Взвешенная оценка: берём максимум по трём независимым сигналам,
        но с ограничением сверху, чтобы не паниковать.
        """
        signals = [
            robot_pct / 100,
            short["ratio"] * 0.8,
            max(night - 0.15, 0) * 1.2,
        ]
        return min(max(signals), 0.85)

# engine/bid_optimizer.py
"""
Скриптовый оптимизатор ставок. Работает по правилам:
- Снижает ставки на keyword'ах с высоким CPA
- Повышает на тех, что приносят конверсии дешевле целевого CPA
- Не трогает кампании без достаточной статистики
"""
from dataclasses import dataclass


@dataclass
class BidAction:
    keyword_id: int
    keyword: str
    old_bid: float
    new_bid: float
    reason: str
    confidence: float


class BidOptimizer:
    MIN_CLICKS_FOR_DECISION = 20   # Минимум кликов для статзначимости
    MAX_BID_INCREASE_PCT = 25      # Не повышать больше чем на 25% за раз
    MAX_BID_DECREASE_PCT = 40      # Не снижать больше чем на 40%

    def optimize(
        self,
        keywords_stats: list[dict],
        target_cpa: float,
    ) -> list[BidAction]:
        actions: list[BidAction] = []

        for kw in keywords_stats:
            clicks = int(kw.get("clicks", 0))
            if clicks < self.MIN_CLICKS_FOR_DECISION:
                continue

            cost = float(kw.get("cost", 0))
            conversions = int(kw.get("conversions", 0))
            current_bid = float(kw.get("bid", 0))

            if current_bid <= 0:
                continue

            # Нет конверсий — агрессивно снижаем
            if conversions == 0:
                new_bid = current_bid * (1 - self.MAX_BID_DECREASE_PCT / 100)
                actions.append(BidAction(
                    keyword_id=kw["id"],
                    keyword=kw.get("keyword", ""),
                    old_bid=current_bid,
                    new_bid=round(new_bid, 2),
                    reason=f"{clicks} кликов, 0 конверсий — снижаем на {self.MAX_BID_DECREASE_PCT}%",
                    confidence=0.9,
                ))
                continue

            actual_cpa = cost / conversions

            if actual_cpa > target_cpa * 1.3:
                # CPA сильно выше целевого — снижаем
                reduction = min(
                    (actual_cpa - target_cpa) / actual_cpa,
                    self.MAX_BID_DECREASE_PCT / 100
                )
                new_bid = current_bid * (1 - reduction)
                actions.append(BidAction(
                    keyword_id=kw["id"],
                    keyword=kw.get("keyword", ""),
                    old_bid=current_bid,
                    new_bid=round(new_bid, 2),
                    reason=f"CPA {actual_cpa:.0f}₽ выше целевого {target_cpa:.0f}₽ на {(actual_cpa/target_cpa-1)*100:.0f}%",
                    confidence=0.85,
                ))
            elif actual_cpa < target_cpa * 0.7:
                # CPA заметно ниже — есть запас для роста
                increase = min(
                    (target_cpa - actual_cpa) / target_cpa * 0.5,
                    self.MAX_BID_INCREASE_PCT / 100
                )
                new_bid = current_bid * (1 + increase)
                actions.append(BidAction(
                    keyword_id=kw["id"],
                    keyword=kw.get("keyword", ""),
                    old_bid=current_bid,
                    new_bid=round(new_bid, 2),
                    reason=f"CPA {actual_cpa:.0f}₽ ниже цели — есть запас для масштабирования",
                    confidence=0.75,
                ))

        return actions

# engine/hypothesis_engine.py
"""
Генератор гипотез на основе правил и статистики.
AI вызывается ОПЦИОНАЛЬНО для улучшения формулировок.
"""
from dataclasses import dataclass
from datetime import datetime


@dataclass
class Hypothesis:
    code: str
    type: str
    title: str
    description: str
    expected_impact_pct: float
    required_budget: float
    duration_days: int
    priority: int
    success_metric: str
    rollback_rule: str


class HypothesisEngine:
    def generate(
        self,
        campaigns: list[dict],
        bot_report: dict,
        kpi: dict,
    ) -> list[Hypothesis]:
        hypotheses: list[Hypothesis] = []

        # ─── Гипотеза: минус-слова по бот-паттернам ────────
        if bot_report.get("bot_percentage", 0) > 15:
            hypotheses.append(Hypothesis(
                code="H001_negative_keywords",
                type="negative_keywords",
                title="Добавление минус-слов по бот-паттернам",
                description=(
                    "Собрать стоп-слова из поисковых запросов с коротким "
                    "временем на сайте и добавить как минус-слова"
                ),
                expected_impact_pct=15.0,
                required_budget=0,
                duration_days=7,
                priority=95,
                success_metric="Снижение bot_percentage на 30%+",
                rollback_rule="Если CR упадёт более чем на 10% — откат",
            ))

        # ─── Гипотеза: временной таргетинг ─────────────────
        if kpi.get("conversion_hours_concentrated", False):
            hypotheses.append(Hypothesis(
                code="H002_time_targeting",
                type="schedule_test",
                title="Концентрация показов в конверсионные часы",
                description=(
                    "Повысить ставки на +30% в часы с макс. CR, "
                    "снизить на -40% в часы с минимальной конверсией"
                ),
                expected_impact_pct=20.0,
                required_budget=kpi.get("daily_budget", 0) * 7,
                duration_days=7,
                priority=85,
                success_metric="Снижение CPA на 15%+",
                rollback_rule="Если impressions упадут > 40% — откат",
            ))

        # ─── Гипотеза: тест автостратегии ──────────────────
        for camp in campaigns:
            if (camp.get("strategy") == "MANUAL"
                    and camp.get("conversions_30d", 0) >= 10):
                hypotheses.append(Hypothesis(
                    code=f"H003_auto_strategy_{camp['id']}",
                    type="bid_test",
                    title=f"Автостратегия для «{camp['name']}»",
                    description=(
                        "Перевести на автоматическую стратегию "
                        "«Оптимизация конверсий» с целевым CPA"
                    ),
                    expected_impact_pct=18.0,
                    required_budget=camp.get("daily_budget", 0) * 14,

"""
Базовый SEO-анализ сайта без AI.
Проверяет: мета-теги, скорость загрузки, mobile, h1-h6, robots.txt
"""
import aiohttp
import re
from dataclasses import dataclass


@dataclass
class SEOIssue:
    code: str
    severity: str
    title: str
    description: str
    recommendation: str


@dataclass
class SEOReport:
    url: str
    score: int                 # 0-100
    issues: list[SEOIssue]
    meta: dict


class SEOAnalyzer:
    async def analyze(self, url: str) -> SEOReport:
        issues: list[SEOIssue] = []
        meta: dict = {}

        async with aiohttp.ClientSession() as s:
            try:
                async with s.get(url, timeout=15) as resp:
                    html = await resp.text()
                    load_time = resp.headers.get("X-Response-Time")
            except Exception as e:
                return SEOReport(url=url, score=0, issues=[SEOIssue(
                    code="unreachable",
                    severity="critical",
                    title="Сайт недоступен",
                    description=str(e),
                    recommendation="Проверьте работу хостинга",
                )], meta={})

        # Title
        title_match = re.search(r"<title>(.*?)</title>", html, re.IGNORECASE | re.DOTALL)
        title = title_match.group(1).strip() if title_match else ""
        meta["title"] = title
        if not title:
            issues.append(SEOIssue(
                "no_title", "critical", "Отсутствует <title>",
                "На странице нет тега title",
                "Добавьте уникальный title длиной 50-60 символов",
            ))
        elif len(title) > 70:
            issues.append(SEOIssue(
                "long_title", "medium", "Длинный title",
                f"Длина title: {len(title)} символов",
                "Сократите до 50-60 символов",
            ))

        # Description
        desc_match = re.search(
            r'<meta\s+name=["\']description["\']\s+content=["\'](.*?)["\']',
            html, re.IGNORECASE,
        )
        description = desc_match.group(1) if desc_match else ""
        meta["description"] = description
        if not description:
            issues.append(SEOIssue(
                "no_description", "high", "Нет meta description",
                "Отсутствует описание страницы",
                "Добавьте description длиной 140-160 символов",
            ))

        # H1
        h1_count = len(re.findall(r"<h1[\s>]", html, re.IGNORECASE))
        meta["h1_count"] = h1_count
        if h1_count == 0:
            issues.append(SEOIssue(
                "no_h1", "high", "Нет H1-заголовка",
                "На странице отсутствует главный заголовок",
                "Добавьте один заголовок H1 с ключевым запросом",
            ))
        elif h1_count > 1:
            issues.append(SEOIssue(
                "multiple_h1", "medium", f"Множественные H1 ({h1_count})",
                "На странице несколько H1-заголовков",
                "Оставьте только один H1",
            ))

        # Viewport (мобильная версия)
        if 'name="viewport"' not in html.lower():
            issues.append(SEOIssue(
                "no_viewport", "high", "Нет mobile viewport",
                "Сайт не адаптирован для мобильных",
                "Добавьте <meta name='viewport' content='width=device-width, initial-scale=1'>",
            ))

        # Open Graph
        if 'property="og:' not in html.lower():
            issues.append(SEOIssue(
                "no_og", "low", "Нет Open Graph",
                "Отсутствуют OG-теги для соцсетей",
                "Добавьте og:title, og:description, og:image",
            ))

        # Расчёт score
        severity_weight = {"critical": 25, "high": 15, "medium": 7, "low": 3}
        penalty = sum(severity_weight.get(i.severity, 0) for i in issues)
        score = max(0, 100 - penalty)

        return SEOReport(url=url, score=score, issues=issues, meta=meta)

    BASE = "https://openrouter.ai/api/v1/chat/completions"

    def __init__(self):

        self.model = settings.AI_MODEL or FREE_MODELS[0]

        self,
        system: str,
        user: str,
        temperature: float = 0.3,
        max_tokens: int = 1500,
    ) -> str:

            "model": self.model,
            "messages": [

                {"role": "user", "content": user},
            ],
            "temperature": temperature,
            "max_tokens": max_tokens,

            "Content-Type": "application/json",

        models_queue = [self.model] + [m for m in FREE_MODELS if m != self.model]

                        self.BASE, json=payload, headers=headers, timeout=60

                            continue

"""

Используется универсальная ссылка СБП (без привязки к конкретному банку).

import qrcode
from io import BytesIO
import uuid
from datetime import datetime, timedelta


class SBPPayment:
    def __init__(self, merchant_id: str, account: str):
        self.merchant_id = merchant_id
        self.account = account

    def create_payment(
        self,
        amount: float,
        user_id: int,
        description: str = "Пополнение баланса AI Marketer",
    ) -> dict:
        """
        Создаёт платёж и возвращает:
        - payment_id (для отслеживания)
        - sbp_url (универсальная СБП-ссылка)
        - qr_png (QR-код в виде PNG-байтов)
        """
        payment_id = str(uuid.uuid4())

        # Универсальная СБП-ссылка (поддерживается всеми банками РФ)
        # Формат: https://qr.nspk.ru/<ID>?type=02&bank=<bank>&sum=<amount>&cur=RUB&crc=<crc>
        sbp_url = (
            f"https://qr.nspk.ru/{self.merchant_id}"
            f"?type=02&sum={int(amount * 100)}&cur=RUB"
            f"&purpose={description}&ref={payment_id}"
        )

        # Генерируем QR
        qr = qrcode.QRCode(
            version=1,
            error_correction=qrcode.constants.ERROR_CORRECT_H,
            box_size=10,
            border=2,
        )
        qr.add_data(sbp_url)
        qr.make(fit=True)
        img = qr.make_image(fill_color="#FFFFFF", back_color="#1C1C1E")

        buf = BytesIO()
        img.save(buf, format="PNG")
        qr_png = buf.getvalue()

        return {
            "payment_id": payment_id,
            "amount": amount,
            "sbp_url": sbp_url,
            "qr_png": qr_png,
            "expires_at": (datetime.utcnow() + timedelta(hours=1)).isoformat(),
        }

"""
Валидация Telegram WebApp initData.
Каждый запрос с фронта содержит initData — проверяем подпись.
"""
import hmac
import hashlib
from urllib.parse import unquote, parse_qsl
from fastapi import Header, HTTPException
from core.config import settings


def verify_init_data(init_data: str) -> dict:
    try:
        parsed = dict(parse_qsl(unquote(init_data)))
        received_hash = parsed.pop("hash")
        data_check = "\n".join(
            f"{k}={v}" for k, v in sorted(parsed.items())
        )
        secret = hmac.new(
            b"WebAppData", settings.BOT_TOKEN.encode(), hashlib.sha256
        ).digest()
        calc_hash = hmac.new(
            secret, data_check.encode(), hashlib.sha256
        ).hexdigest()
        if calc_hash != received_hash:
            raise ValueError("bad signature")
        return parsed
    except Exception as e:
        raise HTTPException(401, f"Invalid initData: {e}")


async def tg_user(x_telegram_init_data: str = Header(...)) -> dict:

      if (!tg) return
      tg.MainButton.setText(text)
      tg.MainButton.onClick(cb)
      tg.MainButton.show()
    },
    hideMainButton: () => tg?.MainButton?.hide(),

    showBackButton: (cb) => {
      if (!tg) return
      tg.BackButton.onClick(cb)
      tg.BackButton.show()
    },
    hideBackButton: () => tg?.BackButton?.hide(),
  }
}

  const h = tg?.HapticFeedback

  return {
    light:   () => h?.impactOccurred?.('light'),
    medium:  () => h?.impactOccurred?.('medium'),
    heavy:   () => h?.impactOccurred?.('heavy'),
    rigid:   () => h?.impactOccurred?.('rigid'),
    soft:    () => h?.impactOccurred?.('soft'),
    success: () => h?.notificationOccurred?.('success'),
    warning: () => h?.notificationOccurred?.('warning'),
    error:   () => h?.notificationOccurred?.('error'),
    select:  () => h?.selectionChanged?.(),
  }
}

// webapp/src/composables/useApi.js
const API_BASE = import.meta.env.VITE_API_BASE || '/api'

export function useApi() {
  const initData = window.Telegram?.WebApp?.initData || ''

  async function request(path, options = {}) {
    const res = await fetch(`${API_BASE}${path}`, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        'X-Telegram-Init-Data': initData,
        ...(options.headers || {}),
      },
    })
    if (!res.ok) {
      const err = await res.json().catch(() => ({}))
      throw new Error(err.detail || `HTTP ${res.status}`)
    }
    return res.json()
  }

  return {
    get:  (p)      => request(p),
    post: (p, b)   => request(p, { method: 'POST', body: JSON.stringify(b) }),
    put:  (p, b)   => request(p, { method: 'PUT',  body: JSON.stringify(b) }),
    del:  (p)      => request(p, { method: 'DELETE' }),
  }
}

<script setup>
import { ref, computed, onUnmounted } from 'vue'
import { useRouter } from 'vue-router'
import GlassCard from '../components/GlassCard.vue'
import GlassButton from '../components/GlassButton.vue'
import { useApi } from '../composables/useApi'
import { useHaptics } from '../composables/useHaptics'
import { useTelegram } from '../composables/useTelegram'

const router = useRouter()
const api = useApi()
const haptics = useHaptics()
const { tg } = useTelegram()

const balance = ref(0)
const amount = ref(5000)
const customAmount = ref(null)
const presets = [1000, 3000, 5000, 10000, 25000, 50000]
const loading = ref(false)
const paymentData = ref(null)
const qrDataUrl = ref('')
let pollInterval = null

const finalAmount = computed(() => customAmount.value || amount.value)

function formatMoney(v) {
  return new Intl.NumberFormat('ru-RU').format(v || 0)
}

function selectAmount(a) {
  haptics.select()
  amount.value = a
  customAmount.value = null
}

async function createPayment() {
  loading.value = true
  haptics.medium()
  try {
    const res = await api.post('/payments/create', {
      amount: finalAmount.value,
    })
    paymentData.value = res
    qrDataUrl.value = `data:image/png;base64,${res.qr_png_base64}`
    startPolling(res.payment_id)
  } catch (e) {
    haptics.error()
    tg?.showAlert?.(e.message)
  } finally {
    loading.value = false
  }
}

function startPolling(paymentId) {
  pollInterval = setInterval(async () => {
    try {
      const s = await api.get(`/payments/${paymentId}/status`)
      if (s.status === 'paid') {
        haptics.success()
        tg?.showAlert?.(`Баланс пополнен на ${formatMoney(finalAmount.value)} ₽`)
        clearInterval(pollInterval)
        router.push('/')
      }
    } catch {}
  }, 3000)
}

function closeSheet() {
  haptics.light()
  clearInterval(pollInterval)
  paymentData.value = null
}

function openSBPLink() {
  tg?.openLink?.(paymentData.value.sbp_url) || window.open(paymentData.value.sbp_url)
}

function back() { router.back() }

onUnmounted(() => clearInterval(pollInterval))
</script>
